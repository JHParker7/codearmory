    http://{{ include "codearmory.fullname" . }}-conductor:{{ .Values.conductor.port }}

  Port-forward for local access:
    kubectl port-forward svc/{{ include "codearmory.fullname" . }}-conductor {{ .Values.conductor.port }}:{{ .Values.conductor.port }} -n {{ .Release.Namespace }}
{{- end }}

Before production use, make sure you have set:
  - gatekeeper.secret.gatekeeperServices   (service auth keys)
  - registry.secret.adminKey / readKey
  - blueprints.secret.encryptionKey        (openssl rand -hex 32)
  - blueprints.secret.gatekeeperServiceKey
  - conductor.secret.registryReadKey
  - conductor.secret.conductorForwardKey
  - forge.secret.conductorForwardKey
  - forge.secret.gatekeeperServiceKey
  - postgresql.auth.postgresPassword       (change from default)
